cmake_minimum_required(VERSION 3.21)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND NOT DEFINED ENV{MACOSX_DEPLOYMENT_TARGET})
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Oldest supported macOS version")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/src/qt/buildinfo.h" version_line
     REGEX "inline constexpr char version\\[\\] = \"[0-9.]+\";")
string(REGEX MATCH "[0-9]+\\.[0-9]+(\\.[0-9]+)?" release_version "${version_line}")
project(CrumbWriter VERSION ${release_version} LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(HUNSPELL REQUIRED IMPORTED_TARGET hunspell)

if(WIN32)
    set(storage_backend src/storage-windows.c)
else()
    set(storage_backend src/storage-posix.c)
endif()
add_library(crumbcore STATIC src/crumb.c src/spacing.c src/storage.c ${storage_backend} src/draft.c)
target_include_directories(crumbcore PUBLIC src)
target_link_libraries(crumbcore PUBLIC PkgConfig::GIO)
# An object library keeps the Qt resource initializer, including in the GUI tests.
file(GLOB editor_sources CONFIGURE_DEPENDS "src/qt/*.cpp")
list(REMOVE_ITEM editor_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/qt/main.cpp")
add_library(crumbgui OBJECT ${editor_sources} data/crumbwriter.qrc)
target_include_directories(crumbgui PUBLIC src/qt)
target_compile_definitions(crumbgui PUBLIC QT_NO_KEYWORDS)
target_link_libraries(crumbgui PUBLIC crumbcore Qt6::Widgets Qt6::Network PkgConfig::HUNSPELL)

qt_add_executable(crumbwriter src/qt/main.cpp)
target_link_libraries(crumbwriter PRIVATE crumbgui)
set(notices LICENSE CrumbWriter.txt)
if(WIN32)
    enable_language(RC)
    set(CRUMB_VERSION_PATCH 0)
    if(PROJECT_VERSION_PATCH)
        set(CRUMB_VERSION_PATCH ${PROJECT_VERSION_PATCH})
    endif()
    configure_file(packaging/windows/crumbwriter.rc.in crumbwriter.rc @ONLY)
    target_sources(crumbwriter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/crumbwriter.rc")
    set_target_properties(crumbwriter PROPERTIES WIN32_EXECUTABLE TRUE)
    install(TARGETS crumbwriter RUNTIME DESTINATION .)
    install(FILES ${notices} DESTINATION .)
elseif(APPLE)
    set_target_properties(crumbwriter PROPERTIES
        OUTPUT_NAME "Crumb Writer" MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in"
        MACOSX_BUNDLE_BUNDLE_NAME "Crumb Writer" MACOSX_BUNDLE_GUI_IDENTIFIER org.thecrumbs.CrumbWriter
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
    set(mac_icon "${CMAKE_CURRENT_BINARY_DIR}/CrumbWriter.icns")
    file(GLOB icon_sources CONFIGURE_DEPENDS "data/icons/*/apps/org.thecrumbs.CrumbWriter.png")
    add_custom_command(OUTPUT "${mac_icon}"
        COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/icon.sh"
            "${CMAKE_CURRENT_SOURCE_DIR}/data/icons" "${mac_icon}"
        DEPENDS ${icon_sources} packaging/macos/icon.sh VERBATIM)
    target_sources(crumbwriter PRIVATE ${notices} "${mac_icon}")
    set_source_files_properties(${notices} "${mac_icon}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    install(TARGETS crumbwriter BUNDLE DESTINATION .)
else()
    include(GNUInstallDirs)
    install(TARGETS crumbwriter RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES ${notices} DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/crumbwriter)
    install(FILES data/org.thecrumbs.CrumbWriter.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(DIRECTORY data/icons/ DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor)
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(suite IN ITEMS crumb storage)
        add_executable(test-${suite} tests/test-${suite}.c)
        target_link_libraries(test-${suite} PRIVATE crumbcore)
        add_test(NAME ${suite} COMMAND test-${suite})
    endforeach()
    foreach(suite IN ITEMS model documentstore bookmarks editor search navigation synonyms spelling window vi resize
                           symbols synonymwriting appearance layout)
        add_executable(test-${suite} tests/qt/test-${suite}.cpp)
        target_link_libraries(test-${suite} PRIVATE crumbgui)
        add_test(NAME ${suite} COMMAND test-${suite})
    endforeach()
endif()
